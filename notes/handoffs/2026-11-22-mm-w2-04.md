## 1. Summary of Changes

Implemented `DP-1` Discretionary Prompts Engine as specified in `docs/playbook-design/04-discretionary-prompts-integration.md` and `BOARD.md` §1b (`MM-W2-04`):
- Defined core types and data structures: `ReasonChip`, `DeviationKind`, `PromptTrigger`, `ReviewPrompt`, `PromptLibraryV1`, `PendingAsk`.
- Implemented `seedPromptLibrary(userId)` creating stable `seed:*` IDs for all 7 `CallKind`s (`skip`, `flip`, `cut`, `early_take`, `stop_here`, `hold`, `size`) and shared reasons, always populating `user_id` (Invariant 7).
- Implemented `detectDeviations(trade: TradeRecord, options?: DeviationOptions): DeviationKind[]` reading `exit_price`, `planned_target_price`, `initial_stop_price`, `stop_loss_price`, `stop_moves`, and `hasWidenedStop`, covering:
  - `exit_before_target_loss` (Cut, including 0-tick scratches per `OD-DP-07`)
  - `exit_before_target_profit` (Early take)
  - `exit_beyond_target` (Hold)
  - `stop_widened` & `stop_tightened` (with pre-R-10 null initial stop guards)
  - `stop_set` vs `no_stop`
  - `size_off_normal` & `against_signal`
  - Slippage tolerance edges (default 1 tick) and exit suppression via `exit_reason` (`target`, `stop`, `trail`).
- Implemented `pendingAsks(trade, library, existingCalls?, options?): PendingAsk[]`:
  - Returns ordered, unanswered prompts whose trigger matches detected deviations or `always`.
  - Prioritizes deviation-triggered prompts over `always` prompts, preserving library order.
  - Excludes already-answered kinds and prompts, permanently dismissed prompts (`not_a_call`), and retired prompts.
  - Enforces at most one ask per `CallKind`.
  - Supports optional `maxAsks` budget slicing.
- Created full unit test suite in `lib/__tests__/discretionary-prompts.test.ts` covering 24 comprehensive test cases.

export type { ReasonChip, ReasonFamily, DeviationKind };

export type PromptTrigger =
  | { on: "always"; when?: DeviationKind }
  | { on: "deviation"; deviations: DeviationKind[] }
  | { on: "manual" };

export interface ReviewPrompt {
  id: string;
  user_id: string;
  seed_id: string | null;
  seed_version: number | null;
  answer_shape: "call" | "reflection";
  kind: CallKind | null;
  primary: boolean;
  question: string;
  reasons: ReasonChip[];
  include_shared_reasons: boolean;
  ask_conviction: boolean;
  trigger: PromptTrigger;
  scope: "global" | "playbook";
  retired: boolean;
}

export interface PromptLibraryV1 {
  schema: "mogmine.prompt_library.v1";
  user_id: string;
  seeds_version: number;
  shared_reasons: ReasonChip[];
  prompts: ReviewPrompt[];
}

export interface PendingAsk {
  prompt: ReviewPrompt;
  deviation: DeviationKind | null;
  trade_id?: string;
}

export function defaultTriggerForKind(kind: CallKind): PromptTrigger;

export function detectDeviations(trade: TradeRecord, options?: DeviationOptions): DeviationKind[];
export function pendingAsks(trade: TradeRecord, library: PromptLibraryV1, existingCalls?: DiscretionCall[] | { calls?: DiscretionCall[]; dismissals?: PromptDismissal[] }, options?: PendingAsksOptions): PendingAsk[];
